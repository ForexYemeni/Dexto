     1→'use client'
     2→
     3→import { useEffect, useState } from 'react'
     4→import { useI18n } from '@/hooks/use-i18n'
     5→import { useAuthStore, useUIStore } from '@/lib/store'
     6→import { motion, AnimatePresence } from 'framer-motion'
     7→import {
     8→  Pickaxe, Clock, TrendingUp, Zap, Award, Gem, Crown, Diamond,
     9→  Check, Loader2, AlertCircle, Play, Lock, Timer, Sparkles, ArrowDownToLine,
    10→} from 'lucide-react'
    11→import { formatCurrency, formatMeccaTime } from '@/lib/time-utils'
    12→import { useToast } from '@/hooks/use-toast'
    13→
    14→interface MiningData {
    15→  plans: any[]
    16→  activeSessions: any[]
    17→  history: any[]
    18→  balance: number
    19→}
    20→
    21→export function MiningView() {
    22→  const { t, locale, isRTL } = useI18n()
    23→  const { user } = useAuthStore()
    24→  const setView = useUIStore((s) => s.setView)
    25→  const { toast } = useToast()
    26→  const [data, setData] = useState<MiningData | null>(null)
    27→  const [loading, setLoading] = useState(true)
    28→  const [startingPlan, setStartingPlan] = useState<string | null>(null)
    29→  const [selectedPlan, setSelectedPlan] = useState<any | null>(null)
    30→  const [investmentAmount, setInvestmentAmount] = useState<number>(0)
    31→  const [showModal, setShowModal] = useState(false)
    32→
    33→  const fetchData = async () => {
    34→    try {
    35→      const res = await fetch('/api/mining')
    36→      const json = await res.json()
    37→      if (!res.ok) {
    38→        console.error('Mining API error:', json)
    39→        // Set fallback data so UI doesn't stay empty
    40→        setData({ plans: [], activeSessions: [], history: [], balance: user?.balance ?? 0 })
    41→        return
    42→      }
    43→      setData(json)
    44→      if (json.balance !== undefined && user) {
    45→        useAuthStore.getState().updateUser({ balance: json.balance })
    46→      }
    47→    } catch (e) {
    48→      console.error(e)
    49→      // Set fallback data
    50→      setData({ plans: [], activeSessions: [], history: [], balance: user?.balance ?? 0 })
    51→    } finally {
    52→      setLoading(false)
    53→    }
    54→  }
    55→
    56→  useEffect(() => {
    57→    fetchData()
    58→    const interval = setInterval(fetchData, 15000)
    59→    return () => clearInterval(interval)
    60→  }, [])
    61→
    62→  const openStartModal = (plan: any) => {
    63→    setSelectedPlan(plan)
    64→    setInvestmentAmount(plan.minInvestment)
    65→    setShowModal(true)
    66→  }
    67→
    68→  const handleStartMining = async () => {
    69→    if (!selectedPlan) return
    70→    if (investmentAmount < selectedPlan.minInvestment || investmentAmount > selectedPlan.maxInvestment) {
    71→      toast({
    72→        variant: 'destructive',
    73→        title: '❌ ' + t('error'),
    74→        description: `${t('minInvestment')}: ${selectedPlan.minInvestment} USDT`,
    75→      })
    76→      return
    77→    }
    78→
    79→    setStartingPlan(selectedPlan.id)
    80→    try {
    81→      const res = await fetch('/api/mining', {
    82→        method: 'POST',
    83→        headers: { 'Content-Type': 'application/json' },
    84→        body: JSON.stringify({
    85→          action: 'subscribe',
    86→          planId: selectedPlan.id,
    87→          investmentAmount,
    88→        }),
    89→      })
    90→      const data = await res.json()
    91→      if (!res.ok) {
    92→        const errMap: Record<string, string> = {
    93→          insufficient_balance: t('insufficientBalance'),
    94→          already_active_for_plan: locale === 'ar' ? 'مشترك بالفعل في هذه الخطة' : 'Already subscribed to this plan',
    95→          invalid_amount: t('error'),
    96→          plan_not_available: locale === 'ar' ? 'الخطة غير متاحة' : 'Plan not available',
    97→        }
    98→        toast({
    99→          variant: 'destructive',
   100→          title: '❌ ' + t('error'),
   101→          description: errMap[data.error] || t('error'),
   102→        })
   103→        return
   104→      }
   105→      toast({
   106→        variant: 'success',
   107→        title: '✅ ' + (locale === 'ar' ? 'تم الاشتراك بنجاح!' : 'Subscribed successfully!'),
   108→        description: locale === 'ar'
   109→          ? `فعّل التعدين يومياً لربح ${formatCurrency(data.session.dailyProfit, locale)} USDT/يوم`
   110→          : `Activate mining daily to earn ${formatCurrency(data.session.dailyProfit, locale)} USDT/day`,
   111→      })
   112→      setShowModal(false)
   113→      fetchData()
   114→    } catch (e) {
   115→      toast({ variant: 'destructive', title: '❌ ' + t('error'), description: t('error') })
   116→    } finally {
   117→      setStartingPlan(null)
   118→    }
   119→  }
   120→
   121→  // Activate daily mining
   122→  const handleActivateMining = async (sessionId: string) => {
   123→    try {
   124→      const res = await fetch('/api/mining', {
   125→        method: 'POST',
   126→        headers: { 'Content-Type': 'application/json' },
   127→        body: JSON.stringify({ action: 'activate', sessionId }),
   128→      })
   129→      const data = await res.json()
   130→      if (!res.ok) {
   131→        const errMap: Record<string, string> = {
   132→          already_mining: locale === 'ar' ? 'التعدين مفعّل بالفعل' : 'Mining already activated',
   133→          plan_completed: locale === 'ar' ? 'اكتملت الخطة' : 'Plan completed',
   134→          session_not_found: locale === 'ar' ? 'الجلسة غير موجودة' : 'Session not found',
   135→        }
   136→        toast({
   137→          variant: 'destructive',
   138→          title: '❌ ' + t('error'),
   139→          description: errMap[data.error] || t('error'),
   140→        })
   141→        return
   142→      }
   143→      toast({
   144→        variant: 'success',
   145→        title: '✅ ' + (locale === 'ar' ? 'تم تفعيل التعدين!' : 'Mining activated!'),
   146→        description: locale === 'ar' ? 'بدأ التعدين لمدة 24 ساعة' : 'Mining started for 24 hours',
   147→      })
   148→      fetchData()
   149→    } catch (e) {
   150→      toast({ variant: 'destructive', title: '❌ ' + t('error') })
   151→    }
   152→  }
   153→
   154→  if (loading || !data) {
   155→    return (
   156→      <div className="space-y-4">
   157→        <div className="h-32 glass rounded-2xl animate-pulse" />
   158→        <div className="grid md:grid-cols-2 gap-4">
   159→          {Array.from({ length: 4 }).map((_, i) => (
   160→            <div key={i} className="h-64 glass rounded-2xl animate-pulse" />
   161→          ))}
   162→        </div>
   163→      </div>
   164→    )
   165→  }
   166→
   167→  const planIcons: Record<string, React.ReactNode> = {
   168→    pickaxe: <Pickaxe className="w-6 h-6" />,
   169→    silver: <Award className="w-6 h-6" />,
   170→    gold: <Crown className="w-6 h-6" />,
   171→    diamond: <Diamond className="w-6 h-6" />,
   172→  }
   173→
   174→  return (
   175→    <div className="space-y-6" dir={isRTL ? 'rtl' : 'ltr'}>
   176→      {/* Active Mining Section */}
   177→      {data.activeSessions.length > 0 && (
   178→        <div>
   179→          <h2 className="text-lg font-bold text-white mb-3 flex items-center gap-2">
   180→            <Zap className="w-5 h-5 text-amber-400" />
   181→            {t('activeMining')}
   182→          </h2>
   183→          <div className="grid md:grid-cols-2 gap-4">
   184→            {data.activeSessions.map((session) => (
   185→              <ActiveMiningCard
   186→                key={session.id}
   187→                session={session}
   188→                onActivate={() => handleActivateMining(session.id)}
   189→              />
   190→            ))}
   191→          </div>
   192→        </div>
   193→      )}
   194→
   195→      {/* Mining Plans */}
   196→      <div>
   197→        <h2 className="text-lg font-bold text-white mb-3 flex items-center gap-2">
   198→          <Pickaxe className="w-5 h-5 text-blue-400" />
   199→          {t('miningPlans')}
   200→        </h2>
   201→        <div className="grid md:grid-cols-2 gap-4">
   202→          {data.plans.map((plan, i) => {
   203→            const isActive = data.activeSessions.some((s) => s.planId === plan.id)
   204→            return (
   205→              <motion.div
   206→                key={plan.id}
   207→                initial={{ opacity: 0, y: 20 }}
   208→                animate={{ opacity: 1, y: 0 }}
   209→                transition={{ delay: i * 0.1 }}
   210→                className="premium-card glass rounded-2xl p-6 relative overflow-hidden"
   211→                style={{ borderColor: `${plan.color}40` }}
   212→              >
   213→                {/* Background glow */}
   214→                <div
   215→                  className="absolute -top-12 -right-12 w-40 h-40 rounded-full blur-3xl opacity-20"
   216→                  style={{ background: plan.color }}
   217→                />
   218→
   219→                <div className="relative">
   220→                  {/* Plan header */}
   221→                  <div className="flex items-start justify-between mb-4">
   222→                    <div className="flex items-center gap-3">
   223→                      <div
   224→                        className="w-12 h-12 rounded-xl flex items-center justify-center text-white"
   225→                        style={{ background: `linear-gradient(135deg, ${plan.color}, ${plan.color}80)` }}
   226→                      >
   227→                        {planIcons[plan.icon] || <Gem className="w-6 h-6" />}
   228→                      </div>
   229→                      <div>
   230→                        <h3 className="text-base font-bold text-white">
   231→                          {locale === 'ar' ? plan.nameAr : plan.name}
   232→                        </h3>
   233→                        <p className="text-xs text-white/40">
   234→                          {locale === 'ar' ? plan.descriptionAr : plan.description}
   235→                        </p>
   236→                      </div>
   237→                    </div>
   238→                    {isActive && (
   239→                      <span className="text-[10px] px-2 py-1 rounded-full bg-green-500/20 text-green-400 flex items-center gap-1">
   240→                        <span className="w-1.5 h-1.5 rounded-full bg-green-400 animate-pulse" />
   241→                        {locale === 'ar' ? 'يعمل' : 'Running'}
   242→                      </span>
   243→                    )}
   244→                  </div>
   245→
   246→                  {/* Plan stats */}
   247→                  <div className="grid grid-cols-2 gap-3 mb-4">
   248→                    <div className="glass rounded-xl p-3">
   249→                      <p className="text-[10px] text-white/40 mb-1">{t('dailyProfit')}</p>
   250→                      <p className="text-lg font-bold text-green-400">
   251→                        {(plan.dailyProfitRate * 100).toFixed(0)}%
   252→                      </p>
   253→                    </div>
   254→                    <div className="glass rounded-xl p-3">
   255→                      <p className="text-[10px] text-white/40 mb-1">{locale === 'ar' ? 'المدة' : 'Duration'}</p>
   256→                      <p className="text-lg font-bold text-white">
   257→                        {plan.totalDays || 7} {locale === 'ar' ? 'أيام' : 'days'}
   258→                      </p>
   259→                    </div>
   260→                    <div className="glass rounded-xl p-3">
   261→                      <p className="text-[10px] text-white/40 mb-1">{t('minInvestment')}</p>
   262→                      <p className="text-sm font-bold text-white">
   263→                        {formatCurrency(plan.minInvestment, locale)}
   264→                      </p>
   265→                    </div>
   266→                    <div className="glass rounded-xl p-3">
   267→                      <p className="text-[10px] text-white/40 mb-1">{t('maxInvestment')}</p>
   268→                      <p className="text-sm font-bold text-white">
   269→                        {formatCurrency(plan.maxInvestment, locale)}
   270→                      </p>
   271→                    </div>
   272→                  </div>
   273→
   274→                  {/* Subscribe button */}
   275→                  <button
   276→                    onClick={() => openStartModal(plan)}
   277→                    disabled={isActive || !plan.isActive}
   278→                    className="w-full py-3 rounded-xl font-semibold text-white transition-all flex items-center justify-center gap-2 disabled:opacity-50 disabled:cursor-not-allowed"
   279→                    style={{
   280→                      background: isActive
   281→                        ? 'rgba(255,255,255,0.05)'
   282→                        : `linear-gradient(135deg, ${plan.color}, ${plan.color}80)`,
   283→                    }}
   284→                  >
   285→                    {isActive ? (
   286→                      <>
   287→                        <Timer className="w-4 h-4" />
   288→                        {locale === 'ar' ? 'مشترك بالفعل' : 'Subscribed'}
   289→                      </>
   290→                    ) : !plan.isActive ? (
   291→                      <>
   292→                        <Lock className="w-4 h-4" />
   293→                        {locale === 'ar' ? 'غير متاحة' : 'Unavailable'}
   294→                      </>
   295→                    ) : (
   296→                      <>
   297→                        <Sparkles className="w-4 h-4" />
   298→                        {locale === 'ar' ? 'اشترك في الخطة' : 'Subscribe to Plan'}
   299→                      </>
   300→                    )}
   301→                  </button>
   302→                  {/* Show balance status */}
   303→                  {!isActive && plan.isActive && (
   304→                    <div className="mt-2 text-center">
   305→                      {data.balance >= plan.minInvestment ? (
   306→                        <p className="text-[10px] text-green-400">
   307→                          {locale === 'ar'
   308→                            ? `رصيدك: ${formatCurrency(data.balance, locale)} USDT ✓`
   309→                            : `Balance: ${formatCurrency(data.balance, locale)} USDT ✓`}
   310→                        </p>
   311→                      ) : (
   312→                        <button
   313→                          onClick={() => setView('deposit')}
   314→                          className="text-[10px] text-amber-400 hover:text-amber-300 transition-colors"
   315→                        >
   316→                          {locale === 'ar'
   317→                            ? `رصيدك ${formatCurrency(data.balance, locale)} USDT - اضغط للإيداع`
   318→                            : `Balance ${formatCurrency(data.balance, locale)} USDT - Click to deposit`}
   319→                        </button>
   320→                      )}
   321→                    </div>
   322→                  )}
   323→                </div>
   324→              </motion.div>
   325→            )
   326→          })}
   327→        </div>
   328→      </div>
   329→
   330→      {/* Mining History */}
   331→      <div>
   332→        <h2 className="text-lg font-bold text-white mb-3 flex items-center gap-2">
   333→          <Clock className="w-5 h-5 text-purple-400" />
   334→          {t('miningHistory')}
   335→        </h2>
   336→        <div className="glass rounded-2xl p-6">
   337→          {data.history.length === 0 ? (
   338→            <div className="text-center py-12">
   339→              <Pickaxe className="w-12 h-12 mx-auto text-white/10 mb-3" />
   340→              <p className="text-sm text-white/40">{t('noMiningHistory')}</p>
   341→            </div>
   342→          ) : (
   343→            <div className="space-y-2 max-h-96 overflow-y-auto custom-scroll">
   344→              {data.history.map((h) => (
   345→                <div key={h.id} className="flex items-center justify-between py-3 border-b border-white/5 last:border-0">
   346→                  <div className="flex items-center gap-3">
   347→                    <div className="w-10 h-10 rounded-xl bg-green-500/10 flex items-center justify-center text-green-400">
   348→                      <Check className="w-5 h-5" />
   349→                    </div>
   350→                    <div>
   351→                      <p className="text-sm font-medium text-white">{h.planName}</p>
   352→                      <p className="text-[10px] text-white/40">
   353→                        {formatMeccaTime(h.startedAt, locale)} → {formatMeccaTime(h.completedAt, locale)}
   354→                      </p>
   355→                    </div>
   356→                  </div>
   357→                  <div className="text-end">
   358→                    <p className="text-sm font-bold text-green-400 tabular-nums">
   359→                      +{formatCurrency(h.profitAmount, locale)}
   360→                    </p>
   361→                    <p className="text-[10px] text-white/40">
   362→                      {formatCurrency(h.investmentAmount, locale)} USDT
   363→                    </p>
   364→                  </div>
   365→                </div>
   366→              ))}
   367→            </div>
   368→          )}
   369→        </div>
   370→      </div>
   371→
   372→      {/* Start mining modal */}
   373→      <AnimatePresence>
   374→        {showModal && selectedPlan && (
   375→          <motion.div
   376→            initial={{ opacity: 0 }}
   377→            animate={{ opacity: 1 }}
   378→            exit={{ opacity: 0 }}
   379→            className="fixed inset-0 bg-black/80 backdrop-blur-sm z-50 flex items-center justify-center p-4"
   380→            onClick={() => setShowModal(false)}
   381→          >
   382→            <motion.div
   383→              initial={{ scale: 0.9, y: 20 }}
   384→              animate={{ scale: 1, y: 0 }}
   385→              exit={{ scale: 0.9, y: 20 }}
   386→              onClick={(e) => e.stopPropagation()}
   387→              className="glass-strong rounded-3xl p-6 w-full max-w-md"
   388→            >
   389→              <div className="flex items-center gap-3 mb-6">
   390→                <div
   391→                  className="w-12 h-12 rounded-xl flex items-center justify-center text-white"
   392→                  style={{ background: `linear-gradient(135deg, ${selectedPlan.color}, ${selectedPlan.color}80)` }}
   393→                >
   394→                  {planIcons[selectedPlan.icon] || <Gem className="w-6 h-6" />}
   395→                </div>
   396→                <div>
   397→                  <h3 className="text-lg font-bold text-white">
   398→                    {locale === 'ar' ? selectedPlan.nameAr : selectedPlan.name}
   399→                  </h3>
   400→                  <p className="text-xs text-white/40">
   401→                    {t('dailyProfit')}: {(selectedPlan.dailyProfitRate * 100).toFixed(0)}%
   402→                  </p>
   403→                </div>
   404→              </div>
   405→
   406→              <div className="space-y-4">
   407→                {/* Balance */}
   408→                <div className="glass rounded-xl p-3 flex items-center justify-between">
   409→                  <span className="text-xs text-white/60">{t('availableBalance')}</span>
   410→                  <span className={`text-sm font-bold tabular-nums ${(user?.balance ?? 0) < selectedPlan.minInvestment ? 'text-amber-400' : 'text-green-400'}`}>
   411→                    {formatCurrency(user?.balance ?? 0, locale)} USDT
   412→                  </span>
   413→                </div>
   414→
   415→                {/* Insufficient balance warning */}
   416→                {(user?.balance ?? 0) < selectedPlan.minInvestment && (
   417→                  <div className="glass rounded-xl p-4 bg-amber-500/10 border border-amber-500/30">
   418→                    <p className="text-xs text-amber-400 mb-3 flex items-center gap-2">
   419→                      <AlertCircle className="w-4 h-4" />
   420→                      {locale === 'ar'
   421→                        ? `رصيدك غير كافٍ. تحتاج ${formatCurrency(selectedPlan.minInvestment - (user?.balance ?? 0), locale)} USDT إضافية`
   422→                        : `Insufficient balance. Need ${formatCurrency(selectedPlan.minInvestment - (user?.balance ?? 0), locale)} USDT more`}
   423→                    </p>
   424→                    <button
   425→                      onClick={() => { setShowModal(false); setView('deposit') }}
   426→                      className="w-full py-2.5 rounded-xl bg-gradient-to-r from-amber-500 to-orange-500 text-white text-sm font-semibold flex items-center justify-center gap-2 hover:scale-[1.02] transition-transform"
   427→                    >
   428→                      <ArrowDownToLine className="w-4 h-4" />
   429→                      {locale === 'ar' ? 'اذهب للإيداع' : 'Go to Deposit'}
   430→                    </button>
   431→                  </div>
   432→                )}
   433→
   434→                {/* Investment amount input */}
   435→                <div>
   436→                  <label className="text-xs text-white/60 mb-1.5 block">{t('investmentAmount')}</label>
   437→                  <input
   438→                    type="number"
   439→                    value={investmentAmount}
   440→                    onChange={(e) => setInvestmentAmount(Number(e.target.value))}
   441→                    min={selectedPlan.minInvestment}
   442→                    max={selectedPlan.maxInvestment}
   443→                    className="w-full bg-white/5 border border-white/10 rounded-xl py-3 px-4 text-white text-lg font-bold focus:outline-none focus:border-blue-500/50"
   444→                  />
   445→                  <div className="flex justify-between text-[10px] text-white/40 mt-1">
   446→                    <span>{t('min')}: {formatCurrency(selectedPlan.minInvestment, locale)}</span>
   447→                    <span>{t('max')}: {formatCurrency(selectedPlan.maxInvestment, locale)}</span>
   448→                  </div>
   449→                </div>
   450→
   451→                {/* Expected profit - daily + total */}
   452→                <div className="glass rounded-xl p-4 bg-gradient-to-br from-green-500/10 to-green-600/5">
   453→                  <div className="flex items-center justify-between mb-2">
   454→                    <span className="text-xs text-white/60 flex items-center gap-1">
   455→                      <TrendingUp className="w-3.5 h-3.5 text-green-400" />
   456→                      {locale === 'ar' ? 'الربح اليومي' : 'Daily Profit'}
   457→                    </span>
   458→                    <span className="text-lg font-bold text-green-400 tabular-nums">
   459→                      +{formatCurrency(investmentAmount * selectedPlan.dailyProfitRate, locale)} USDT
   460→                    </span>
   461→                  </div>
   462→                  <div className="flex items-center justify-between mb-2 pt-2 border-t border-white/10">
   463→                    <span className="text-xs text-white/60">
   464→                      {locale === 'ar' ? `إجمالي الأرباح (${selectedPlan.totalDays || 7} أيام)` : `Total Profit (${selectedPlan.totalDays || 7} days)`}
   465→                    </span>
   466→                    <span className="text-lg font-bold text-purple-400 tabular-nums">
   467→                      +{formatCurrency(investmentAmount * selectedPlan.dailyProfitRate * (selectedPlan.totalDays || 7), locale)} USDT
   468→                    </span>
   469→                  </div>
   470→                  <div className="flex items-center gap-2 pt-2 border-t border-white/10">
   471→                    <Lock className="w-3 h-3 text-amber-400" />
   472→                    <p className="text-[10px] text-white/50">
   473→                      {locale === 'ar'
   474→                        ? `رأس المال مقفل لمدة ${selectedPlan.totalDays || 7} أيام - تسحب الأرباح يومياً`
   475→                        : `Capital locked for ${selectedPlan.totalDays || 7} days - withdraw profits daily`}
   476→                    </p>
   477→                  </div>
   478→                </div>
   479→
   480→                {/* Action buttons */}
   481→                <div className="flex gap-2">
   482→                  <button
   483→                    onClick={() => setShowModal(false)}
   484→                    className="flex-1 py-3 rounded-xl glass text-white text-sm font-medium hover:bg-white/10 transition-colors"
   485→                  >
   486→                    {t('cancel')}
   487→                  </button>
   488→                  <button
   489→                    onClick={handleStartMining}
   490→                    disabled={startingPlan === selectedPlan.id}
   491→                    className="flex-1 py-3 rounded-xl bg-gradient-to-r from-blue-500 to-purple-500 text-white text-sm font-semibold hover:scale-[1.02] transition-transform disabled:opacity-50 flex items-center justify-center gap-2"
   492→                  >
